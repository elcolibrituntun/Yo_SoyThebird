local EveryFoundWeapon = {}

local function DeepScanWeapons() 
local TempList = {} 
for _, Name in ipairs(FullMM2WeaponDatabase) do 
table.insert(TempList, Name) 
end 
local Weapons = ReplicatedStorage:FindFirstChild("Weapons") or ReplicatedStorage:FindFirstChild("Items") 
if Weapons then 
for _, Folder in ipairs(Weapons:GetChildren()) do 
for _, Item in ipairs(Folder:GetChildren()) do 
if not table.find(TempList, Item.Name) then 
table.insert(TempList, Item.Name) 
end 
end 
end 
end 
EveryFoundWeapon = TempList 
end

local function FindPlayerPartial(Text) 
if Text  "" then return nil end 
local LowerText = string.lower(Text) 
for _, P in ipairs(Players:GetPlayers()) do 
if string.find(string.lower(P.Name), LowerText, 1, true) or string.find(string.lower(P.DisplayName), LowerText, 1, true) then 
return P 
end 
end 
return nil 
end

local LocalPlayer = Players.LocalPlayer 
local TraditionalEvent = LocalPlayer:WaitForChild("ClientRemotes", 5) and LocalPlayer.ClientRemotes:WaitForChild("KnifeThrow", 5)

local function GetCobaltEvent() 
local Character = LocalPlayer.Character 
if Character then 
local Knife = Character:FindFirstChild("Knife") 
if Knife then 
local Events = Knife:FindFirstChild("Events") 
if Events then return Events:FindFirstChild("KnifeThrown") end 
end 
end 
return nil 
end

local function FireKnifeFromCharacter(Character) 
if not Character then return end 
local Torso = Character:FindFirstChild("UpperTorso") or Character:FindFirstChild("Torso") or Character:FindFirstChild("HumanoidRootPart") 
if not Torso then return end 
local Target = GetWorkspacePart() or Instance.new("Part") 
local CobaltEvent = GetCobaltEvent()

-- Itera dinámicamente según la cantidad definida en KnivesPerSecond
for k = 1, KnivesPerSecond do
	if IsRandomSpamActive then
		for i = 1, WeaponsToSpamAmount do
			task.spawn(function()
				local Grados = math.random() * 6.28318
				local ConvX = math.cos(Grados)
				local ConvZ = math.sin(Grados)
				local Offset = SafeDistanceActive and Vector3.new(0, 3, 0) or Vector3.new(0, 0, 0)
				local P_Inicio = Torso.Position + Vector3.new(ConvX * 0.05, 0, ConvZ * 0.05) + Offset
				local P_Destino = Torso.Position + Vector3.new(ConvX * 600, 0, ConvZ * 600) + Offset
				local RandomWeapon = EveryFoundWeapon[math.random(1, #EveryFoundWeapon)]
				
				-- Eventos por defecto
				if TraditionalEvent then
					TraditionalEvent:InvokeServer(CFrame.new(P_Inicio, P_Destino), P_Destino, RandomWeapon, Target)
				end
				if CobaltEvent then
					CobaltEvent:FireServer(CFrame.new(P_Inicio, P_Destino), CFrame.new(P_Destino, Vector3.new(0,1,0)))
				end
				
				-- NUEVO: Ejecuta también todos los RemoteEvents personalizados añadidos por ti
				for _, customEvent in ipairs(CustomRemoteEvents) do
					pcall(function()
						customEvent:FireServer(CFrame.new(P_Inicio, P_Destino), P_Destino, RandomWeapon, Target)
					end)
				end
			end)
		end
	else
		task.spawn(function()
			local Grados = math.random() * 6.28318
			local ConvX = math.cos(Grados)
			local ConvZ = math.sin(Grados)
			local Offset = SafeDistanceActive and Vector3.new(0, 3, 0) or Vector3.new(0, 0, 0)
			local P_Inicio = Torso.Position + Vector3.new(ConvX * 0.05, 0, ConvZ * 0.05) + Offset
			local P_Destino = Torso.Position + Vector3.new(ConvX * 600, 0, ConvZ * 600) + Offset
			
			-- Eventos por defecto
			if TraditionalEvent then
				TraditionalEvent:InvokeServer(CFrame.new(P_Inicio, P_Destino), P_Destino, ExperimentalKnifeName, Target)
			end
			if CobaltEvent then
				CobaltEvent:FireServer(CFrame.new(P_Inicio, P_Destino), CFrame.new(P_Destino, Vector3.new(0,1,0)))
			end
			
			-- NUEVO: Ejecuta también todos los RemoteEvents personalizados añadidos por ti
			for _, customEvent in ipairs(CustomRemoteEvents) do
				pcall(function()
					customEvent:FireServer(CFrame.new(P_Inicio, P_Destino), P_Destino, ExperimentalKnifeName, Target)
				end)
			end
		end)
	end
end

end

function GetWorkspacePart() 
local Parts = {} 
for _, Object in ipairs(workspace:GetDescendants()) do 
if Object:IsA("BasePart") and Object.CanQuery then table.insert(Parts, Object) end 
end 
if #Parts > 0 then return Parts[math.random(1, #Parts)] end 
return nil 
end
